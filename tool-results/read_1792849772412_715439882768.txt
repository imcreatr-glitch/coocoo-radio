     1→     1→"use client";
     2→     2→
     3→     3→/**
     4→     4→ * CooCoo Radio — Live Studio (voice room)
     5→     5→ * Live podcast / discussion room:
     6→     6→ * - Listener: join by name, hear the current speaker (MSE webm/opus playback),
     7→     7→ *   raise hand to ask for the mic
     8→     8→ * - Speaker (when admin grants you the mic): mic streams via MediaRecorder
     9→     9→ * - Admin (host console): start/end room, take mic, grant mic to raised hands
    10→    10→ * Transport: socket.io mini-service voice-service (:3004) via Caddy gateway.
    11→    11→ */
    12→    12→import { useCallback, useEffect, useRef, useState } from "react";
    13→    13→import { io, type Socket } from "socket.io-client";
    14→    14→import {
    15→    15→  AudioLines,
    16→    16→  Crown,
    17→    17→  Hand,
    18→    18→  Headphones,
    19→    19→  Loader2,
    20→    20→  Mic,
    21→    21→  MicOff,
    22→    22→  PhoneOff,
    23→    23→  PowerOff,
    24→    24→  Radio,
    25→    25→  Volume2,
    26→    26→} from "lucide-react";
    27→    27→import { Badge } from "@/components/ui/badge";
    28→    28→import { Button } from "@/components/ui/button";
    29→    29→import { Input } from "@/components/ui/input";
    30→    30→import { SectionTitle } from "@/components/site/schedule";
    31→    31→import { getToken } from "@/lib/client-auth";
    32→    32→
    33→    33→interface LiveMember {
    34→    34→  id: string;
    35→    35→  name: string;
    36→    36→  isAdmin: boolean;
    37→    37→  raised: boolean;
    38→    38→}
    39→    39→
    40→    40→interface LiveState {
    41→    41→  active: boolean;
    42→    42→  startedAt: number | null;
    43→    43→  hostName: string | null;
    44→    44→  speakerId: string | null;
    45→    45→  speakerName: string | null;
    46→    46→  sessionId: number;
    47→    47→  members: LiveMember[];
    48→    48→  queue: string[];
    49→    49→}
    50→    50→
    51→    51→interface MseHandle {
    52→    52→  mse: MediaSource;
    53→    53→  sb: SourceBuffer | null;
    54→    54→  url: string;
    55→    55→  session: number;
    56→    56→}
    57→    57→
    58→    58→/**
    59→    59→ * Audio wire frame: [4B sessionId BE][4B seq BE][1B flags] + payload
    60→    60→ * flags bit 0x01 = payload is the webm/EBML session header.
    61→    61→ */
    62→    62→const FRAME_HEADER = 9;
    63→    63→const FLAG_HEADER = 0x01;
    64→    64→
    65→    65→const NAME_KEY = "coocoo_live_name";
    66→    66→const ADMIN_EVENT = "coocoo:admin-changed";
    67→    67→
    68→    68→export default function LiveStudio() {
    69→    69→  const socketRef = useRef<Socket | null>(null);
    70→    70→  const audioRef = useRef<HTMLAudioElement | null>(null);
    71→    71→  const mseRef = useRef<MseHandle | null>(null);
    72→    72→  const pendingRef = useRef<ArrayBuffer[]>([]);
    73→    73→  const lastSeqRef = useRef(0);
    74→    74→  const recorderRef = useRef<MediaRecorder | null>(null);
    75→    75→  const streamRef = useRef<MediaStream | null>(null);
    76→    76→  const joinedRef = useRef(false);
    77→    77→  const nameRef = useRef("");
    78→    78→
    79→    79→  const [state, setState] = useState<LiveState | null>(null);
    80→    80→  const [connected, setConnected] = useState(false);
    81→    81→  const [joined, setJoined] = useState(false);
    82→    82→  const [name, setName] = useState("");
    83→    83→  const [myId, setMyId] = useState<string | null>(null);
    84→    84→  const [isAdmin, setIsAdmin] = useState(false);
    85→    85→  const [volume, setVolume] = useState(90);
    86→    86→  const [err, setErr] = useState("");
    87→    87→  const [micErr, setMicErr] = useState("");
    88→    88→  const [speaking, setSpeaking] = useState(false);
    89→    89→  const [mseOk, setMseOk] = useState(true);
    90→    90→  const [recOk, setRecOk] = useState(true);
    91→    91→
    92→    92→  const st = state;
    93→    93→  const imSpeaker = !!myId && st?.active && st.speakerId === myId;
    94→    94→
    95→    95→  /* ---------- playback (MSE) ---------- */
    96→    96→
    97→    97→  const teardownPlayback = useCallback(() => {
    98→    98→    pendingRef.current = [];
    99→    99→    const p = mseRef.current;
   100→   100→    mseRef.current = null;
   101→   101→    if (p) {
   102→   102→      try {
   103→   103→        if (p.mse.readyState === "open") p.mse.endOfStream();
   104→   104→      } catch {
   105→   105→        /* ignore */
   106→   106→      }
   107→   107→      URL.revokeObjectURL(p.url);
   108→   108→    }
   109→   109→  }, []);
   110→   110→
   111→   111→  const pump = useCallback(() => {
   112→   112→    const p = mseRef.current;
   113→   113→    if (!p || !p.sb || p.sb.updating) return;
   114→   114→    const next = pendingRef.current.shift();
   115→   115→    if (!next) return;
   116→   116→    try {
   117→   117→      p.sb.appendBuffer(next);
   118→   118→    } catch {
   119→   119→      /* buffer error -> session will be rebuilt on next voice:session */
   120→   120→    }
   121→   121→  }, []);
   122→   122→
   123→   123→  const startPlaybackSession = useCallback(
   124→   124→    (sessionId: number, headerBytes: Uint8Array) => {
   125→   125→      const audio = audioRef.current;
   126→   126→      if (!audio || !mseOk) return;
   127→   127→      teardownPlayback();
   128→   128→      lastSeqRef.current = 0;
   129→   129→
   130→   130→      const mse = new MediaSource();
   131→   131→      const url = URL.createObjectURL(mse);
   132→   132→      mseRef.current = { mse, sb: null, url, session: sessionId };
   133→   133→
   134→   134→      mse.addEventListener(
   135→   135→        "sourceopen",
   136→   136→        () => {
   137→   137→          const p = mseRef.current;
   138→   138→          if (!p || p.session !== sessionId) return;
   139→   139→          let sb: SourceBuffer;
   140→   140→          try {
   141→   141→            sb = mse.addSourceBuffer('audio/webm; codecs="opus"');
   142→   142→          } catch {
   143→   143→            setMseOk(false);
   144→   144→            return;
   145→   145→          }
   146→   146→          p.sb = sb;
   147→   147→          try {
   148→   148→            sb.mode = "sequence";
   149→   149→          } catch {
   150→   150→            /* some browsers may not allow switching mode */
   151→   151→          }
   152→   152→          sb.addEventListener("updateend", pump);
   153→   153→          pendingRef.current = [headerBytes];
   154→   154→          pump();
   155→   155→          audio.play().catch(() => {});
   156→   156→        },
   157→   157→        { once: true }
   158→   158→      );
   159→   159→      audio.src = url;
   160→   160→      audio.play().catch(() => {});
   161→   161→    },
   162→   162→    [mseOk, pump, teardownPlayback]
   163→   163→  );
   164→   164→
   165→   165→  /** binary audio frame from the voice service */
   166→   166→  const handleFrame = useCallback(
   167→   167→    (d: ArrayBuffer | Uint8Array) => {
   168→   168→      const u8 = d instanceof Uint8Array ? new Uint8Array(d) : new Uint8Array(d);
   169→   169→      if (u8.byteLength < FRAME_HEADER + 1) return;
   170→   170→      const dv = new DataView(u8.buffer, u8.byteOffset, u8.byteLength);
   171→   171→      const sid = dv.getUint32(0);
   172→   172→      const seq = dv.getUint32(4);
   173→   173→      const flags = dv.getUint8(8);
   174→   174→      const payload = u8.subarray(FRAME_HEADER);
   175→   175→      if (flags & FLAG_HEADER) {
   176→   176→        // new speaker session -> rebuild playback around this header
   177→   177→        startPlaybackSession(sid, payload);
   178→   178→        return;
   179→   179→      }
   180→   180→      const p = mseRef.current;
   181→   181→      if (!p || sid !== p.session) return; // stale or unknown session
   182→   182→      if (seq <= lastSeqRef.current) return; // duplicate / out of order
   183→   183→      lastSeqRef.current = seq;
   184→   184→      pendingRef.current.push(payload);
   185→   185→      pump();
   186→   186→    },
   187→   187→    [pump, startPlaybackSession]
   188→   188→  );
   189→   189→
   190→   190→  // trim old buffered audio so long sessions don't grow memory
   191→   191→  useEffect(() => {
   192→   192→    if (!joined) return;
   193→   193→    const t = window.setInterval(() => {
   194→   194→      const p = mseRef.current;
   195→   195→      const audio = audioRef.current;
   196→   196→      if (!p || !p.sb || p.sb.updating || !audio) return;
   197→   197→      try {
   198→   198→        if (p.mse.readyState === "open" && p.sb.buffered.length >= 1) {
   199→   199→          const start = p.sb.buffered.start(0);
   200→   200→          const cut = audio.currentTime - 15;
   201→   201→          if (cut > start + 2) p.sb.remove(start, cut);
   202→   202→        }
   203→   203→      } catch {
   204→   204→        /* ignore */
   205→   205→      }
   206→   206→    }, 30000);
   207→   207→    return () => window.clearInterval(t);
   208→   208→  }, [joined]);
   209→   209→
   210→   210→  /* ---------- speaking (MediaRecorder) ---------- */
   211→   211→
   212→   212→  const stopSpeaking = useCallback(() => {
   213→   213→    const rec = recorderRef.current;
   214→   214→    recorderRef.current = null;
   215→   215→    if (rec) {
   216→   216→      rec.ondataavailable = null;
   217→   217→      try {
   218→   218→        rec.stop();
   219→   219→      } catch {
   220→   220→        /* ignore */
   221→   221→      }
   222→   222→    }
   223→   223→    const s = streamRef.current;
   224→   224→    streamRef.current = null;
   225→   225→    if (s) s.getTracks().forEach((t) => t.stop());
   226→   226→    setSpeaking(false);
   227→   227→  }, []);
   228→   228→
   229→   229→  const startSpeaking = useCallback(async () => {
   230→   230→    if (recorderRef.current) return;
   231→   231→    setMicErr("");
   232→   232→    try {
   233→   233→      const stream = await navigator.mediaDevices.getUserMedia({
   234→   234→        audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
   235→   235→      });
   236→   236→      streamRef.current = stream;
   237→   237→      const mime = MediaRecorder.isTypeSupported("audio/webm;codecs=opus")
   238→   238→        ? "audio/webm;codecs=opus"
   239→   239→        : "audio/webm";
   240→   240→      const rec = new MediaRecorder(stream, { mimeType: mime, audioBitsPerSecond: 32000 });
   241→   241→      rec.ondataavailable = (e) => {
   242→   242→        if (e.data && e.data.size > 0) {
   243→   243→          // raw binary chunk (top-level) — service wraps it with seq/session header
   244→   244→          e.data
   245→   245→            .arrayBuffer()
   246→   246→            .then((b) => socketRef.current?.emit("voice:chunk", b))
   247→   247→            .catch(() => {});
   248→   248→        }
   249→   249→      };
   250→   250→      rec.onerror = () => stopSpeaking();
   251→   251→      rec.start(1000);
   252→   252→      recorderRef.current = rec;
   253→   253→      setSpeaking(true);
   254→   254→    } catch {
   255→   255→      // mic denied / unavailable -> give the mic back
   256→   256→      socketRef.current?.emit("voice:release");
   257→   257→      setMicErr("Mic permission nahi mili — browser vich mic allow karo te dobara try karo");
   258→   258→    }
   259→   259→  }, [stopSpeaking]);
   260→   260→
   261→   261→  // auto start/stop mic streaming when I become / stop being the speaker
   262→   262→  useEffect(() => {
   263→   263→    if (imSpeaker) {
   264→   264→      void startSpeaking();
   265→   265→    } else {
   266→   266→      stopSpeaking();
   267→   267→    }
   268→   268→  }, [imSpeaker, startSpeaking, stopSpeaking]);
   269→   269→
   270→   270→  /* ---------- socket ---------- */
   271→   271→
   272→   272→  useEffect(() => {
   273→   273→    const s = io("/?XTransformPort=3004", {
   274→   274→      path: "/",
   275→   275→      transports: ["websocket", "polling"],
   276→   276→      reconnection: true,
   277→   277→      reconnectionAttempts: 10,
   278→   278→      reconnectionDelay: 1500,
   279→   279→      timeout: 10000,
   280→   280→    });
   281→   281→    socketRef.current = s;
   282→   282→
   283→   283→    s.on("connect", () => {
   284→   284→      setConnected(true);
   285→   285→      setMyId(s.id ?? null);
   286→   286→      setErr("");
   287→   287→      // re-join after reconnect so presence stays alive
   288→   288→      if (joinedRef.current) {
   289→   289→        s.emit("voice:join", { name: nameRef.current, token: getToken() ?? undefined });
   290→   290→      }
   291→   291→    });
   292→   292→    s.on("disconnect", () => setConnected(false));
   293→   293→    s.on("connect_error", () => setConnected(false));
   294→   294→
   295→   295→    s.on("voice:state", (d: LiveState) => {
   296→   296→      if (d && Array.isArray(d.members)) setState(d);
   297→   297→    });
   298→   298→    s.on("voice:frame", (d: ArrayBuffer | Uint8Array) => handleFrame(d));
   299→   299→    s.on("voice:error", (d: { error?: string }) => setErr(d?.error || "Koi error aaya"));
   300→   300→
   301→   301→    return () => {
   302→   302→      s.disconnect();
   303→   303→      socketRef.current = null;
   304→   304→    };
   305→   305→  }, [handleFrame]);
   306→   306→
   307→   307→  // cleanup media on unmount
   308→   308→  useEffect(() => {
   309→   309→    return () => {
   310→   310→      stopSpeaking();
   311→   311→      teardownPlayback();
   312→   312→    };
   313→   313→  }, [stopSpeaking, teardownPlayback]);
   314→   314→
   315→   315→  /* ---------- support + name + admin detection ---------- */
   316→   316→
   317→   317→  useEffect(() => {
   318→   318→    const mseSupported =
   319→   319→      typeof MediaSource !== "undefined" && MediaSource.isTypeSupported('audio/webm; codecs="opus"');
   320→   320→    setMseOk(mseSupported);
   321→   321→    setRecOk(
   322→   322→      typeof MediaRecorder !== "undefined" &&
   323→   323→        (MediaRecorder.isTypeSupported("audio/webm;codecs=opus") ||
   324→   324→          MediaRecorder.isTypeSupported("audio/webm"))
   325→   325→    );
   326→   326→    const t = setTimeout(() => {
   327→   327→      const saved = window.localStorage.getItem(NAME_KEY);
   328→   328→      if (saved) {
   329→   329→        setName(saved);
   330→   330→        nameRef.current = saved;
   331→   331→      }
   332→   332→      setIsAdmin(!!getToken());
   333→   333→    }, 0);
   334→   334→    const onAdmin = () => setIsAdmin(!!getToken());
   335→   335→    window.addEventListener(ADMIN_EVENT, onAdmin);
   336→   336→    return () => {
   337→   337→      window.clearTimeout(t);
   338→   338→      window.removeEventListener(ADMIN_EVENT, onAdmin);
   339→   339→    };
   340→   340→  }, []);
   341→   341→
   342→   342→  useEffect(() => {
   343→   343→    if (audioRef.current) audioRef.current.volume = volume / 100;
   344→   344→  }, [volume]);
   345→   345→
   346→   346→  /* ---------- actions ---------- */
   347→   347→
   348→   348→  const emitJoin = useCallback(() => {
   349→   349→    const s = socketRef.current;
   350→   350→    if (!s) return;
   351→   351→    s.emit("voice:join", { name: nameRef.current, token: getToken() ?? undefined });
   352→   352→  }, []);
   353→   353→
   354→   354→  const join = useCallback(() => {
   355→   355→    const n = name.trim() || "Sarota";
   356→   356→    nameRef.current = n;
   357→   357→    joinedRef.current = true;
   358→   358→    window.localStorage.setItem(NAME_KEY, n);
   359→   359→    emitJoin();
   360→   360→    setJoined(true);
   361→   361→    setErr("");
   362→   362→    audioRef.current?.play().catch(() => {});
   363→   363→  }, [name, emitJoin]);
   364→   364→
   365→   365→  const hostAction = useCallback((event: string, extra: Record<string, unknown> = {}) => {
   366→   366→    const token = getToken();
   367→   367→    if (!token) {
   368→   368→      setErr("Eh action sirf admin layi hai — pehla admin panel vich login karo");
   369→   369→      return;
   370→   370→    }
   371→   371→    socketRef.current?.emit(event, { token, ...extra });
   372→   372→    setErr("");
   373→   373→  }, []);
   374→   374→
   375→   375→  const leave = useCallback(() => {
   376→   376→    socketRef.current?.emit("voice:release");
   377→   377→    socketRef.current?.emit("voice:unraise");
   378→   378→    joinedRef.current = false;
   379→   379→    setJoined(false);
   380→   380→    stopSpeaking();
   381→   381→    teardownPlayback();
   382→   382→  }, [stopSpeaking, teardownPlayback]);
   383→   383→
   384→   384→  /* ---------- derived ---------- */
   385→   385→
   386→   386→  const members = st?.members ?? [];
   387→   387→  const onlineCount = members.length;
   388→   388→  const liveSeconds = st?.startedAt ? Math.max(0, Math.floor((Date.now() - st.startedAt) / 1000)) : 0;
   389→   389→  const liveLabel =
   390→   390→    liveSeconds > 0
   391→   391→      ? `${Math.floor(liveSeconds / 3600)}:${String(Math.floor((liveSeconds % 3600) / 60)).padStart(2, "0")}:${String(liveSeconds % 60).padStart(2, "0")}`
   392→   392→      : null;
   393→   393→  const iAmInQueue = !!myId && st?.queue.includes(myId);
   394→   394→
   395→   395→  /* ---------- render ---------- */
   396→   396→
   397→   397→  return (
   398→   398→    <section
   399→   399→      id="live"
   400→   400→      className={`border-b border-amber-400/10 transition-colors ${
   401→   401→        st?.active ? "bg-[oklch(0.185_0.03_30)]" : "bg-[oklch(0.165_0.012_60)]"
   402→   402→      }`}
   403→   403→      data-testid="live-studio"
   404→   404→    >
   405→   405→      <div className="mx-auto max-w-7xl px-4 py-14 sm:px-6">
   406→   406→        <SectionTitle
   407→   407→          title="ਲਾਈਵ ਸਟੂਡੀਓ • Live Voice Room"
   408→   408→          sub="Live podcast discussion — mic naal judo! Admin room start karega, tusi sunn sakde ho te hath utha ke discussion vich hissa le sakde ho"
   409→   409→        />
   410→   410→
   411→   411→        {/* status strip */}
   412→   412→        <div
   413→   413→          className="mb-6 flex flex-wrap items-center gap-3 rounded-2xl border border-amber-400/20 bg-black/30 px-4 py-3"
   414→   414→          data-testid="live-status"
   415→   415→        >
   416→   416→          {st?.active ? (
   417→   417→            <>
   418→   418→              <Badge className="live-dot gap-1.5 border-0 bg-red-600 px-3 py-1 text-xs font-bold text-white">
   419→   419→                <Radio className="h-3.5 w-3.5" /> ON AIR
   420→   420→              </Badge>
   421→   421→              <span className="text-sm font-semibold text-amber-100">
   422→   422→                {st.speakerName ? (
   423→   423→                  <>
   424→   424→                    <MicOff className="mr-1 inline h-4 w-4 text-amber-400" />
   425→   425→                    <span className="font-gur">{st.speakerName}</span> mic te bol rahe ne
   426→   426→                  </>
   427→   427→                ) : (
   428→   428→                  "Mic khali hai — hath chuko te bolna shuru karo!"
   429→   429→                )}
   430→   430→              </span>
   431→   431→              {liveLabel && (
   432→   432→                <span className="ml-auto font-mono text-xs tabular-nums text-amber-300/80">
   433→   433→                  ⏱ {liveLabel}
   434→   434→                </span>
   435→   435→              )}
   436→   436→            </>
   437→   437→          ) : (
   438→   438→            <>
   439→   439→              <Badge variant="outline" className="gap-1.5 border-amber-400/30 px-3 py-1 text-xs text-amber-300">
   440→   440→                <PowerOff className="h-3.5 w-3.5" /> STUDIO OFF
   441→   441→              </Badge>
   442→   442→              <span className="font-gur text-sm text-amber-100/70">
   443→   443→                Live room band hai — admin start karega ta ethe aawaz aavegi
   444→   444→              </span>
   445→   445→            </>
   446→   446→          )}
   447→   447→          <Badge
   448→   448→            variant="outline"
   449→   449→            className={`ml-auto gap-1 border-emerald-400/40 bg-emerald-400/10 text-[10px] text-emerald-300 ${
   450→   450→              st?.active ? "" : "ml-0"
   451→   451→            }`}
   452→   452→          >
   453→   453→            <Headphones className="h-3 w-3" /> {onlineCount} vich shamil
   454→   454→          </Badge>
   455→   455→        </div>
   456→   456→
   457→   457→        {!mseOk && (
   458→   458→          <p className="mb-4 rounded-xl border border-red-400/30 bg-red-500/10 px-4 py-2 text-xs text-red-300">
   459→   459→            Tuhada browser live audio playback support nahi karda — Chrome ya Edge vich kholo te maza
   460→   460→            lao.
   461→   461→          </p>
   462→   462→        )}
   463→   463→
   464→   464→        <div className="grid gap-6 lg:grid-cols-[1fr_360px]">
   465→   465→          {/* ---------- left: stage ---------- */}
   466→   466→          <div
   467→   467→            className="flex flex-col overflow-hidden rounded-2xl border border-amber-400/20 bg-[oklch(0.21_0.018_60)]"
   468→   468→            data-testid="live-stage"
   469→   469→          >
   470→   470→            <div className="flex items-center justify-between border-b border-amber-400/15 bg-black/25 px-4 py-3">
   471→   471→              <h3 className="flex items-center gap-2 text-sm font-bold text-amber-100">
   472→   472→                <AudioLines className="h-4.5 w-4.5 text-amber-400" /> Studio Stage
